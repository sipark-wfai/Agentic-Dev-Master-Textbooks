### 2.1 JWT 인증 스토어

```typescript
// apps/frontend/src/lib/stores/auth.store.ts
import { writable, derived } from 'svelte/store';
import { browser } from '$app/environment';
import type { User } from '$lib/types';

interface AuthState {
  user: User | null;
  accessToken: string | null;
  isLoading: boolean;
}

const initialState: AuthState = {
  user: null,
  accessToken: null,
  isLoading: true,
};

function createAuthStore() {
  const { subscribe, set, update } = writable<AuthState>(initialState);

  return {
    subscribe,

    /**
     * 로그인 상태 설정
     */
    setAuth(user: User, accessToken: string) {
      if (browser) {
        // ⚠️ [보안 경고] localStorage에 JWT 저장은 XSS 공격에 취약합니다.
        // 프로덕션에서는 반드시 httpOnly 쿠키 또는 메모리 저장 방식을 사용하세요.
        // 이 코드는 교육 목적의 단순화된 예시입니다. <!-- R4 수정: localStorage XSS 보안 경고 강화 -->
        localStorage.setItem('accessToken', accessToken);
      }
      set({ user, accessToken, isLoading: false });
    },

    /**
     * 로그아웃
     */
    clearAuth() {
      if (browser) {
        localStorage.removeItem('accessToken');
      }
      set({ user: null, accessToken: null, isLoading: false });
    },

    /**
     * 앱 초기화 시 로컬스토리지에서 토큰 복원
     */
    async initialize() {
      if (!browser) {
        update(s => ({ ...s, isLoading: false }));
        return;
      }

      const savedToken = localStorage.getItem('accessToken');
      if (!savedToken) {
        update(s => ({ ...s, isLoading: false }));
        return;
      }

      try {
        // 토큰으로 내 정보 조회
        const response = await fetch('/api/v1/auth/me', {
          headers: { Authorization: `Bearer ${savedToken}` },
        });

        if (response.ok) {
          const { data: user } = await response.json();
          set({ user, accessToken: savedToken, isLoading: false });
        } else {
          // 토큰 만료 시 refresh 시도
          await this.refreshToken();
        }
      } catch {
        update(s => ({ ...s, isLoading: false }));
      }
    },

    /**
     * Access Token 갱신 (Refresh Token 사용)
     */
    async refreshToken(): Promise<boolean> {
      try {
        const response = await fetch('/api/v1/auth/refresh', {
          method: 'POST',
          credentials: 'include', // refresh token은 쿠키로 전송
        });

        if (response.ok) {
          const { data } = await response.json();
          if (browser) {
            localStorage.setItem('accessToken', data.accessToken);
          }
          // 내 정보 다시 조회
          const meResponse = await fetch('/api/v1/auth/me', {
            headers: { Authorization: `Bearer ${data.accessToken}` },
          });
          if (meResponse.ok) {
            const { data: user } = await meResponse.json();
            set({ user, accessToken: data.accessToken, isLoading: false });
            return true;
          }
        }
      } catch {
        // refresh 실패
      }

      this.clearAuth();
      return false;
    },
  };
}

export const authStore = createAuthStore();

// 편의 derived 스토어
export const isLoggedIn = derived(authStore, ($auth) => !!$auth.user);
export const currentUser = derived(authStore, ($auth) => $auth.user);
```

### 2.2 API 클라이언트

```typescript
// apps/frontend/src/lib/api/client.ts
import { authStore } from '$lib/stores/auth.store';
import { get } from 'svelte/store';
import { goto } from '$app/navigation';

const API_BASE = 'http://localhost:3000/api/v1';

interface RequestOptions extends RequestInit {
  skipAuth?: boolean;
}

/**
 * 인증이 포함된 API 요청 헬퍼
 */
async function request<T>(endpoint: string, options: RequestOptions = {}): Promise<T> {
  const { skipAuth = false, ...fetchOptions } = options;

  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
    ...(fetchOptions.headers as Record<string, string>),
  };

  if (!skipAuth) {
    const auth = get(authStore);
    if (auth.accessToken) {
      headers['Authorization'] = `Bearer ${auth.accessToken}`;
    }
  }

  const response = await fetch(`${API_BASE}${endpoint}`, {
    ...fetchOptions,
    headers,
    credentials: 'include',
  });

  // 401: 토큰 만료 → refresh 시도
  if (response.status === 401 && !skipAuth) {
    const refreshed = await authStore.refreshToken();
    if (refreshed) {
      // 재시도
      const auth = get(authStore);
      headers['Authorization'] = `Bearer ${auth.accessToken}`;
      const retryResponse = await fetch(`${API_BASE}${endpoint}`, {
        ...fetchOptions,
        headers,
        credentials: 'include',
      });
      return retryResponse.json();
    } else {
      goto('/auth/login');
      throw new Error('인증이 만료되었습니다');
    }
  }

  if (!response.ok) {
    const error = await response.json();
    throw new Error(error.message || '요청에 실패했습니다');
  }

  return response.json();
}

export const apiClient = {
  get: <T>(endpoint: string, options?: RequestOptions) =>
    request<T>(endpoint, { ...options, method: 'GET' }),

  post: <T>(endpoint: string, body?: unknown, options?: RequestOptions) =>
    request<T>(endpoint, {
      ...options,
      method: 'POST',
      body: JSON.stringify(body),
    }),

  patch: <T>(endpoint: string, body?: unknown, options?: RequestOptions) =>
    request<T>(endpoint, {
      ...options,
      method: 'PATCH',
      body: JSON.stringify(body),
    }),

  delete: <T>(endpoint: string, options?: RequestOptions) =>
    request<T>(endpoint, { ...options, method: 'DELETE' }),
};
```

```typescript
// apps/frontend/src/routes/board/[board_id]/+page.server.ts
import type { PageServerLoad } from './$types';
import { error } from '@sveltejs/kit';

export const load: PageServerLoad = async ({ params, url, fetch }) => {
  const boardId = params.board_id;
  const page = Number(url.searchParams.get('page') || '1');
  const limit = Number(url.searchParams.get('limit') || '20');
  const search = url.searchParams.get('search') || '';

  try {
    const [boardRes, postsRes] = await Promise.all([
      fetch(`http://localhost:3000/api/v1/boards/${boardId}`),
      fetch(
        `http://localhost:3000/api/v1/boards/${boardId}/posts?` +
          new URLSearchParams({ page: String(page), limit: String(limit), search })
      ),
    ]);

    if (!boardRes.ok) {
      throw error(404, '게시판을 찾을 수 없습니다');
    }

    const { data: board } = await boardRes.json();
    const { data: posts, meta } = await postsRes.json();

    return { board, posts, meta, page, search };
  } catch (e) {
    throw error(500, '데이터를 불러오는 데 실패했습니다');
  }
};
```

### 2.4 파일 업로드 드롭존 컴포넌트

```svelte
<!-- apps/frontend/src/lib/components/FileUploadDropzone.svelte -->
<script lang="ts">
  import { createEventDispatcher } from 'svelte';
  import { apiClient } from '$lib/api/client';

  export let boardType: string;
  export let postId: string | undefined = undefined;
  export let maxFiles = 10;
  export let accept = '*';

  const dispatch = createEventDispatcher<{
    uploaded: { id: string; filename: string; url: string; mimeType: string; size: number }[];
    error: string;
  }>();

  let isDragging = false;
  let uploadedFiles: { id: string; filename: string; url: string }[] = [];
  let isUploading = false;

  // 게시판 타입별 파일 제한 설정
  const fileConfig = {
    GALLERY: { accept: 'image/jpeg,image/png,image/gif,image/webp', maxSize: 10 * 1024 * 1024 },
    FILE: { accept: '*', maxSize: 50 * 1024 * 1024 },
    default: { accept: '*', maxSize: 10 * 1024 * 1024 },
  };

  const config = fileConfig[boardType as keyof typeof fileConfig] || fileConfig.default;

  async function handleFiles(files: FileList) {
    if (files.length === 0) return;
    if (uploadedFiles.length + files.length > maxFiles) {
      dispatch('error', `최대 ${maxFiles}개까지 업로드 가능합니다`);
      return;
    }

    isUploading = true;
    const results = [];

    for (const file of Array.from(files)) {
      if (file.size > config.maxSize) {
        dispatch('error', `${file.name}: 파일 크기가 초과됩니다 (최대 ${config.maxSize / 1024 / 1024}MB)`);
        continue;
      }

      const formData = new FormData();
      formData.append('file', file);
      if (postId) formData.append('postId', postId);

      try {
        const result = await fetch('http://localhost:3000/api/v1/files/upload', {
          method: 'POST',
          body: formData,
          headers: {
            // <!-- R3 수정: 컴포넌트에서 localStorage 직접 접근 대신 authStore.$accessToken 사용 권장 -->
            Authorization: `Bearer ${localStorage.getItem('accessToken')}`,
          },
        });
        if (result.ok) {
          const { data } = await result.json();
          uploadedFiles = [...uploadedFiles, data];
          results.push(data);
        }
      } catch (e) {
        dispatch('error', `${file.name} 업로드 실패`);
      }
    }

    dispatch('uploaded', results);
    isUploading = false;
  }

  function handleDragOver(e: DragEvent) {
    e.preventDefault();
    isDragging = true;
  }

  function handleDrop(e: DragEvent) {
    e.preventDefault();
    isDragging = false;
    if (e.dataTransfer?.files) {
      handleFiles(e.dataTransfer.files);
    }
  }

  function removeFile(id: string) {
    uploadedFiles = uploadedFiles.filter(f => f.id !== id);
  }
</script>

<div
  class="border-2 border-dashed rounded-xl p-8 text-center transition
         {isDragging ? 'border-blue-500 bg-blue-50' : 'border-gray-300 hover:border-gray-400'}"
  on:dragover={handleDragOver}
  on:dragleave={() => isDragging = false}
  on:drop={handleDrop}
  role="region"
  aria-label="파일 업로드 영역"
>
  {#if isUploading}
    <p class="text-blue-600">업로드 중...</p>
  {:else}
    <p class="text-gray-500">파일을 여기에 드래그하거나</p>
    <label class="cursor-pointer text-blue-600 hover:underline mt-2 inline-block">
      파일 선택
      <input
        type="file"
        class="hidden"
        accept={config.accept}
        multiple={maxFiles > 1}
        on:change={(e) => e.target?.files && handleFiles(e.target.files)}
      />
    </label>
    <p class="text-xs text-gray-400 mt-1">
      최대 {maxFiles}개, {config.maxSize / 1024 / 1024}MB 이하
    </p>
  {/if}
</div>

{#if uploadedFiles.length > 0}
  <ul class="mt-3 space-y-2">
    {#each uploadedFiles as file}
      <li class="flex items-center justify-between bg-gray-50 rounded px-3 py-2 text-sm">
        <span class="truncate">{file.filename}</span>
        <button on:click={() => removeFile(file.id)} class="text-red-500 ml-2">✕</button>
      </li>
    {/each}
  </ul>
{/if}
```

```typescript
// apps/backend/src/file/file.controller.ts
import {
  Controller, Post, UploadedFile, UseGuards,
  UseInterceptors, Body
} from '@nestjs/common';
import { FileInterceptor } from '@nestjs/platform-express';
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
import { FileService } from './file.service';
import { CurrentUser } from '../auth/decorators/current-user.decorator';
import type { User } from '@prisma/client';

@Controller('files')
@UseGuards(JwtAuthGuard)
export class FileController {
  constructor(private readonly fileService: FileService) {}

  @Post('upload')
  @UseInterceptors(FileInterceptor('file'))
  async uploadFile(
    @UploadedFile() file: Express.Multer.File,
    @Body('postId') postId: string,
    @CurrentUser() user: User,
  ) {
    return this.fileService.upload(file, postId, user.id);
  }
}
```
