## 2. ERD → Prisma 스키마 변환

### 2.1 Agent 지시문

```
"@Docs docs/erd.md
이 ERD를 기반으로 apps/backend/prisma/schema.prisma를 완성해줘.

변환 규칙:
- ERD의 모든 엔티티 → Prisma model
- ERD의 PK → @id @default(cuid())
- ERD의 UK → @unique
- ERD의 FK → @relation
- ERD의 soft delete 컬럼 → deletedAt DateTime?
- 모든 모델에 createdAt @default(now()), updatedAt @updatedAt 추가
- 주요 FK 컬럼에 @@index 추가

완성 후 pnpm prisma validate 실행해줘."
```

### 2.2 완성된 schema.prisma

```prisma
// apps/backend/prisma/schema.prisma

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// 열거형 정의
enum Role {
  SUPER_ADMIN
  BOARD_ADMIN
  USER
  GUEST
}

enum BoardType {
  NOTICE
  FREE
  QNA
  GALLERY
  FILE
  NEWS
  POLL
  SECRET
}

// 사용자 모델
model User {
  id            String         @id @default(cuid())
  email         String         @unique
  name          String
  role          Role           @default(USER)
  passwordHash  String
  posts         Post[]
  comments      Comment[]
  refreshTokens RefreshToken[]
  pollVotes     PollVote[]
  createdAt     DateTime       @default(now())
  updatedAt     DateTime       @updatedAt
  deletedAt     DateTime?

  @@index([email])
  @@index([deletedAt])
  @@index([role])
}

// 게시판 모델
model Board {
  id          String    @id @default(cuid())
  name        String    @unique
  type        BoardType
  description String?   @db.VarChar(200)
  isActive    Boolean   @default(true)
  posts       Post[]
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt
  deletedAt   DateTime?

  @@index([type])
  @@index([isActive])
  @@index([deletedAt])
}

// 게시글 모델
model Post {
  id                 String       @id @default(cuid())
  title              String       @db.VarChar(100)
  content            String       @db.Text
  boardId            String
  authorId           String
  viewCount          Int          @default(0)
  isPinned           Boolean      @default(false)
  externalUrl        String?      // NEWS 게시판용
  acceptedCommentId  String?      // QNA 채택된 댓글
  pollExpiresAt      DateTime?    // POLL 마감일
  pollAllowMultiple  Boolean      @default(false)
  board              Board        @relation(fields: [boardId], references: [id])
  author             User         @relation(fields: [authorId], references: [id])
  comments           Comment[]
  files              File[]
  pollOptions        PollOption[]
  createdAt          DateTime     @default(now())
  updatedAt          DateTime     @updatedAt
  deletedAt          DateTime?

  @@index([boardId])
  @@index([authorId])
  @@index([deletedAt])
  @@index([createdAt])
  @@index([isPinned])
}

// 댓글 모델 (대댓글 포함)
model Comment {
  id        String    @id @default(cuid())
  content   String    @db.VarChar(500)
  postId    String
  authorId  String
  parentId  String?   // null = 최상위 댓글
  post      Post      @relation(fields: [postId], references: [id])
  author    User      @relation(fields: [authorId], references: [id])
  parent    Comment?  @relation("Replies", fields: [parentId], references: [id])
  replies   Comment[] @relation("Replies")
  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@index([postId])
  @@index([authorId])
  @@index([parentId])
}

// 파일 모델
model File {
  id            String   @id @default(cuid())
  filename      String   // 원본 파일명
  url           String   // S3 URL
  mimeType      String
  size          Int      // 바이트 단위
  downloadCount Int      @default(0)
  postId        String
  post          Post     @relation(fields: [postId], references: [id])
  createdAt     DateTime @default(now())

  @@index([postId])
}

// Refresh Token 모델
model RefreshToken {
  id        String   @id @default(cuid())
  token     String   @unique
  userId    String
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  expiresAt DateTime
  createdAt DateTime @default(now())

  @@index([userId])
  @@index([token])
  @@index([expiresAt])
}

// 투표 선택지 모델
model PollOption {
  id        String     @id @default(cuid())
  text      String     @db.VarChar(100)
  postId    String
  order     Int        // 정렬 순서
  post      Post       @relation(fields: [postId], references: [id])
  votes     PollVote[]
  createdAt DateTime   @default(now())

  @@index([postId])
  @@index([postId, order])
}

// 투표 참여 모델
model PollVote {
  id        String     @id @default(cuid())
  userId    String
  optionId  String
  user      User       @relation(fields: [userId], references: [id])
  option    PollOption @relation(fields: [optionId], references: [id])
  createdAt DateTime   @default(now())

  @@unique([userId, optionId])  // 같은 선택지에 중복 투표 방지
  @@index([userId])
  @@index([optionId])
}
```

### 3.2 Controller 코드 패턴

```typescript
// apps/backend/src/board/board.controller.ts
import {
  Controller, Get, Post, Patch, Delete,
  Body, Param, UseGuards, HttpCode, HttpStatus
} from '@nestjs/common';
import { ApiTags, ApiOperation, ApiResponse, ApiBearerAuth } from '@nestjs/swagger';
import { BoardService } from './board.service';
import { CreateBoardDto } from './dto/create-board.dto';
import { UpdateBoardDto } from './dto/update-board.dto';
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
import { RolesGuard } from '../auth/guards/roles.guard';
import { Roles } from '../auth/decorators/roles.decorator';
import { Role } from '@prisma/client';

@ApiTags('게시판')
@Controller('boards')
export class BoardController {
  constructor(private readonly boardService: BoardService) {}

  @Get()
  @ApiOperation({ summary: '게시판 목록 조회 [BOARD-001]' })
  @ApiResponse({ status: 200, description: '게시판 목록 반환' })
  findAll() {
    return this.boardService.findAll();
  }

  @Get(':id')
  @ApiOperation({ summary: '게시판 상세 조회 [BOARD-002]' })
  findOne(@Param('id') id: string) {
    return this.boardService.findOne(id);
  }

  @Post()
  @UseGuards(JwtAuthGuard, RolesGuard)
  @Roles(Role.BOARD_ADMIN, Role.SUPER_ADMIN)
  @ApiBearerAuth()
  @ApiOperation({ summary: '게시판 생성 [BOARD-003] (BOARD_ADMIN+)' })
  @ApiResponse({ status: 201, description: '게시판 생성 성공' })
  @ApiResponse({ status: 403, description: '권한 없음' })
  create(@Body() createBoardDto: CreateBoardDto) {
    return this.boardService.create(createBoardDto);
  }

  @Patch(':id')
  @UseGuards(JwtAuthGuard, RolesGuard)
  @Roles(Role.BOARD_ADMIN, Role.SUPER_ADMIN)
  @ApiBearerAuth()
  @ApiOperation({ summary: '게시판 수정 [BOARD-004] (BOARD_ADMIN+)' })
  update(@Param('id') id: string, @Body() updateBoardDto: UpdateBoardDto) {
    return this.boardService.update(id, updateBoardDto);
  }

  @Delete(':id')
  @UseGuards(JwtAuthGuard, RolesGuard)
  @Roles(Role.SUPER_ADMIN)
  @HttpCode(HttpStatus.NO_CONTENT)
  @ApiBearerAuth()
  @ApiOperation({ summary: '게시판 비활성화 [BOARD-005] (SUPER_ADMIN)' })
  remove(@Param('id') id: string) {
    return this.boardService.remove(id);
  }
}
```

### 3.3 Service 코드 패턴

```typescript
// apps/backend/src/board/board.service.ts
import { Injectable, NotFoundException, ConflictException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { CreateBoardDto } from './dto/create-board.dto';
import { UpdateBoardDto } from './dto/update-board.dto';

@Injectable()
export class BoardService {
  constructor(private readonly prisma: PrismaService) {}

  /**
   * 게시판 목록 조회 [BOARD-001]
   * isActive=true인 게시판만 반환, 게시글 수 포함
   */
  async findAll() {
    const boards = await this.prisma.board.findMany({
      where: { isActive: true, deletedAt: null },
      include: {
        _count: { select: { posts: { where: { deletedAt: null } } } },
      },
      orderBy: { createdAt: 'asc' },
    });
    return { data: boards };
  }

  /**
   * 게시판 단건 조회 [BOARD-002]
   */
  async findOne(id: string) {
    const board = await this.prisma.board.findUnique({
      where: { id, deletedAt: null },
      include: {
        _count: { select: { posts: { where: { deletedAt: null } } } },
      },
    });
    if (!board) {
      throw new NotFoundException(`게시판 ID ${id}를 찾을 수 없습니다`);
    }
    return { data: board };
  }

  /**
   * 게시판 생성 [BOARD-003]
   */
  async create(dto: CreateBoardDto) {
    const existing = await this.prisma.board.findFirst({
      where: { name: dto.name, deletedAt: null },
    });
    if (existing) {
      throw new ConflictException(`'${dto.name}' 게시판이 이미 존재합니다`);
    }

    const board = await this.prisma.board.create({ data: dto });
    return { data: board };
  }

  /**
   * 게시판 수정 [BOARD-004]
   */
  async update(id: string, dto: UpdateBoardDto) {
    await this.findOne(id);  // 존재 확인
    const board = await this.prisma.board.update({
      where: { id },
      data: dto,
    });
    return { data: board };
  }

  /**
   * 게시판 비활성화 [BOARD-005] (소프트 삭제)
   */
  async remove(id: string) {
    await this.findOne(id);  // 존재 확인
    await this.prisma.board.update({
      where: { id },
      data: { deletedAt: new Date(), isActive: false },
    });
  }
}
```

```typescript
// apps/backend/src/board/dto/create-board.dto.ts
import { IsString, IsEnum, IsOptional, IsBoolean, MinLength, MaxLength } from 'class-validator';
import { ApiProperty, ApiPropertyOptional } from '@nestjs/swagger';
import { BoardType } from '@prisma/client';

export class CreateBoardDto {
  @ApiProperty({ description: '게시판 이름 (2~50자)', example: '자유게시판' })
  @IsString()
  @MinLength(2, { message: '게시판 이름은 2자 이상이어야 합니다' })
  @MaxLength(50, { message: '게시판 이름은 50자를 초과할 수 없습니다' })
  name: string;

  @ApiProperty({ description: '게시판 유형', enum: BoardType })
  @IsEnum(BoardType, { message: '올바른 게시판 유형을 선택해주세요' })
  type: BoardType;

  @ApiPropertyOptional({ description: '게시판 설명 (최대 200자)' })
  @IsString()
  @IsOptional()
  @MaxLength(200, { message: '게시판 설명은 200자를 초과할 수 없습니다' })
  description?: string;

  @ApiPropertyOptional({ description: '활성화 여부', default: true })
  @IsBoolean()
  @IsOptional()
  isActive?: boolean = true;
}
```

## 4. RBAC (역할 기반 접근 제어) 구현

### 4.1 RolesGuard와 Roles 데코레이터

```typescript
// apps/backend/src/auth/decorators/roles.decorator.ts
import { SetMetadata } from '@nestjs/common';
import { Role } from '@prisma/client';

export const ROLES_KEY = 'roles';
export const Roles = (...roles: Role[]) => SetMetadata(ROLES_KEY, roles);

// apps/backend/src/auth/guards/roles.guard.ts
import { Injectable, CanActivate, ExecutionContext } from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { Role } from '@prisma/client';
import { ROLES_KEY } from '../decorators/roles.decorator';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const requiredRoles = this.reflector.getAllAndOverride<Role[]>(ROLES_KEY, [
      context.getHandler(),
      context.getClass(),
    ]);

    if (!requiredRoles) return true;

    const { user } = context.switchToHttp().getRequest();
    
    // 역할 계층: SUPER_ADMIN > BOARD_ADMIN > USER > GUEST
    const roleHierarchy = {
      [Role.SUPER_ADMIN]: 4,
      [Role.BOARD_ADMIN]: 3,
      [Role.USER]: 2,
      [Role.GUEST]: 1,
    };

    const userLevel = roleHierarchy[user.role] || 0;
    return requiredRoles.some(role => userLevel >= roleHierarchy[role]);
  }
}
```

## 6. PrismaService 소프트 삭제 미들웨어 완성본

```typescript
// apps/backend/src/prisma/prisma.service.ts
import {
  INestApplication,
  Injectable,
  Logger,
  OnModuleInit,
} from '@nestjs/common';
import { PrismaClient } from '@prisma/client';

@Injectable()
export class PrismaService extends PrismaClient implements OnModuleInit {
  private readonly logger = new Logger(PrismaService.name);

  constructor() {
    super({
      log: [
        { emit: 'event', level: 'query' },
        { emit: 'stdout', level: 'warn' },
        { emit: 'stdout', level: 'error' },
      ],
    });
  }

  async onModuleInit() {
    await this.$connect();
    this.logger.log('데이터베이스 연결 완료');
    this.applySoftDeleteMiddleware();
  }

  private applySoftDeleteMiddleware() {
    this.$use(async (params, next) => {
      // 조회 시 삭제된 항목 제외
      if (params.action === 'findUnique' || params.action === 'findFirst') {
        params.action = params.action === 'findUnique' ? 'findFirst' : params.action;
        if (!params.args.where) params.args.where = {};
        if (params.args.where.deletedAt === undefined) {
          params.args.where.deletedAt = null;
        }
      }

      if (params.action === 'findMany') {
        if (!params.args) params.args = {};
        if (!params.args.where) params.args.where = {};
        if (params.args.where.deletedAt === undefined) {
          params.args.where.deletedAt = null;
        }
      }

      // delete → update (소프트 삭제)
      if (params.action === 'delete') {
        params.action = 'update';
        if (!params.args.data) params.args.data = {};
        params.args.data.deletedAt = new Date();
      }

      if (params.action === 'deleteMany') {
        params.action = 'updateMany';
        if (!params.args.data) params.args.data = {};
        params.args.data.deletedAt = new Date();
      }

      return next(params);
    });
  }

  async enableShutdownHooks(app: INestApplication) {
    process.on('beforeExit', async () => {
      await app.close();
    });
  }
}
```
