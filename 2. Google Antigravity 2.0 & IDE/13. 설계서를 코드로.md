// prisma/schema.prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model User {
  id           Int       @id @default(autoincrement())
  email        String    @unique
  nickname     String
  passwordHash String    @map("password_hash")
  role         UserRole  @default(USER)
  profileImage String?   @map("profile_image")
  createdAt    DateTime  @default(now()) @map("created_at")
  updatedAt    DateTime  @updatedAt @map("updated_at")
  deletedAt    DateTime? @map("deleted_at")

  posts    Post[]
  comments Comment[]

  @@map("users")
}

model Board {
  id          Int       @id @default(autoincrement())
  name        String
  slug        String    @unique
  boardType   BoardType @map("board_type")
  description String?
  isActive    Boolean   @default(true) @map("is_active")
  sortOrder   Int       @default(0) @map("sort_order")
  postsCount  Int       @default(0) @map("posts_count")
  createdAt   DateTime  @default(now()) @map("created_at")

  posts Post[]

  @@map("boards")
}

model Post {
  id               Int         @id @default(autoincrement())
  boardId          Int         @map("board_id")
  authorId         Int?        @map("author_id")
  title            String      @db.VarChar(200)
  content          String
  viewCount        Int         @default(0) @map("view_count")
  commentsCount    Int         @default(0) @map("comments_count")
  isAnswered       Boolean     @default(false) @map("is_answered")
  acceptedAnswerId Int?        @map("accepted_answer_id")
  faqCategory      FaqCategory? @map("faq_category")
  createdAt        DateTime    @default(now()) @map("created_at")
  updatedAt        DateTime    @updatedAt @map("updated_at")
  deletedAt        DateTime?   @map("deleted_at")

  board    Board     @relation(fields: [boardId], references: [id])
  author   User?     @relation(fields: [authorId], references: [id])
  files    File[]
  comments Comment[]

  @@index([boardId, deletedAt])
  @@index([authorId, deletedAt])
  @@map("posts")
}

enum UserRole {
  USER
  ADMIN
}

enum BoardType {
  GENERAL
  GALLERY
  VIDEO
  QNA
  NOTICE
  FAQ
  PRIVATE
  ANONYMOUS
}

### 2.2 게시판 타입별 처리

에이전트가 자동으로 게시판 타입에 따른 분기를 생성한다:

```typescript
// src/posts/posts.service.ts (에이전트 생성)
async create(createPostDto: CreatePostDto, authorId: number | null): Promise<Post> {
  const board = await this.prisma.board.findUnique({
    where: { id: createPostDto.boardId }
  });

  if (!board) throw new NotFoundException('게시판을 찾을 수 없습니다');

  // 게시판 타입별 검증
  switch (board.boardType) {
    case BoardType.GALLERY:
      if (!createPostDto.images?.length) {
        throw new BadRequestException('갤러리 게시판은 이미지가 필수입니다');
      }
      break;
    case BoardType.VIDEO:
      if (!createPostDto.video) {
        throw new BadRequestException('동영상 게시판은 동영상이 필수입니다');
      }
      break;
    case BoardType.NOTICE:
      // ADMIN 권한 검증은 Guard에서 처리
      break;
    case BoardType.ANONYMOUS:
      authorId = null; // 익명 게시판
      break;
  }

  return this.prisma.post.create({
    data: {
      boardId: createPostDto.boardId,
      authorId,
      title: createPostDto.title,
      content: createPostDto.content,
    }
  });
}
```

## 3. 소프트 딜리트 자동화

### 3.1 Prisma v5 $extends() 소프트 딜리트

```
/code "Prisma 서비스에 소프트 딜리트 미들웨어를 추가해줘.
  Prisma v5의 $extends() 패턴을 사용해.
  findMany, findOne은 deletedAt이 null인 것만 조회"
```

```typescript
// src/prisma/prisma.service.ts (에이전트 생성)
import { Injectable, OnModuleInit } from '@nestjs/common';
import { PrismaClient } from '@prisma/client';

@Injectable()
export class PrismaService extends PrismaClient implements OnModuleInit {
  async onModuleInit() {
    await this.$connect();
  }
}

// 소프트 딜리트 확장 버전
export const prismaWithSoftDelete = new PrismaClient().$extends({
  query: {
    $allModels: {
      async findMany({ args, query }) {
        args.where = { ...(args.where || {}), deletedAt: null };
        return query(args);
      },
      async findFirst({ args, query }) {
        args.where = { ...(args.where || {}), deletedAt: null };
        return query(args);
      },
      async findUnique({ args, query }) {
        return query(args); // unique는 소프트 딜리트 필터 제외
      },
    },
  },
});
```

// src/common/dto/pagination.dto.ts (에이전트 생성)
import { IsInt, IsOptional, Max, Min } from 'class-validator';
import { Type } from 'class-transformer';

export class PaginationDto {
  @IsOptional()
  @Type(() => Number)
  @IsInt()
  @Min(1)
  page?: number = 1;

  @IsOptional()
  @Type(() => Number)
  @IsInt()
  @Min(1)
  @Max(100)
  limit?: number = 20;
}

export interface PageMeta {
  total: number;
  page: number;
  limit: number;
  totalPages: number;
  hasNext: boolean;
  hasPrev: boolean;
}
