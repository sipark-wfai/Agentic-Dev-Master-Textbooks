## 2. NestJS 전체 모듈 구조

### 2.1 main.ts 설정

```typescript
// apps/backend/src/main.ts

import { NestFactory } from '@nestjs/core';
import { ValidationPipe } from '@nestjs/common';
import { SwaggerModule, DocumentBuilder } from '@nestjs/swagger';
import { ConfigService } from '@nestjs/config';
import * as cookieParser from 'cookie-parser';
import { AppModule } from './app.module';
import { HttpExceptionFilter } from './common/filters/http-exception.filter';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);
  const config = app.get(ConfigService);

  // Cookie Parser
  app.use(cookieParser());

  // CORS 설정 (허용된 오리진만)
  app.enableCors({
    origin: config.get<string>('FRONTEND_URL', 'http://localhost:5173'),
    credentials: true,
    methods: ['GET', 'POST', 'PUT', 'PATCH', 'DELETE', 'OPTIONS'],
  });

  // 전역 유효성 검증 파이프
  app.useGlobalPipes(
    new ValidationPipe({
      whitelist: true,       // 정의되지 않은 필드 제거
      forbidNonWhitelisted: true,
      transform: true,       // 타입 자동 변환
      transformOptions: { enableImplicitConversion: true },
    }),
  );

  // 전역 예외 필터
  app.useGlobalFilters(new HttpExceptionFilter());

  // Swagger 설정 (개발 환경만)
  if (config.get('NODE_ENV') !== 'production') {
    const swaggerConfig = new DocumentBuilder()
      .setTitle('BoardHub API')
      .setDescription('멀티 게시판 허브 API 문서')
      .setVersion('1.0')
      .addBearerAuth()
      .build();
    const document = SwaggerModule.createDocument(app, swaggerConfig);
    SwaggerModule.setup('api-docs', app, document);
  }

  const port = config.get<number>('PORT', 3000);
  await app.listen(port);
  console.log(`BoardHub Backend 서버 실행 중: http://localhost:${port}`);
  console.log(`Swagger: http://localhost:${port}/api-docs`);
}

bootstrap();
```

## 3. SvelteKit 구현

### 3.1 인증 스토어 (JWT 관리)

```typescript
// apps/frontend/src/lib/stores/auth.ts

import { writable, derived } from 'svelte/store';

interface User {
  id: number;
  email: string;
  nickname: string;
  role: 'ADMIN' | 'MODERATOR' | 'USER';
}

interface AuthState {
  user: User | null;
  accessToken: string | null;
  isLoading: boolean;
}

function createAuthStore() {
  const { subscribe, set, update } = writable<AuthState>({
    user: null,
    accessToken: null,
    isLoading: false,
  });

  return {
    subscribe,

    /** 로그인 상태 설정 */
    setAuth: (user: User, accessToken: string) => {
      update(state => ({ ...state, user, accessToken, isLoading: false }));
    },

    /** 로그아웃 */
    clearAuth: () => {
      set({ user: null, accessToken: null, isLoading: false });
    },

    /** Access Token 갱신 */
    updateToken: (accessToken: string) => {
      update(state => ({ ...state, accessToken }));
    },
  };
}

export const authStore = createAuthStore();

/** 파생 스토어: 로그인 여부 */
export const isAuthenticated = derived(
  authStore,
  $auth => $auth.accessToken !== null
);

/** 파생 스토어: 관리자 여부 */
export const isAdmin = derived(
  authStore,
  $auth => $auth.user?.role === 'ADMIN'
);
```

### 3.2 레이아웃 서버 로드 (+layout.server.ts)

```typescript
// apps/frontend/src/routes/+layout.server.ts

import type { LayoutServerLoad } from './$types';
import { api } from '$lib/api/client';

export const load: LayoutServerLoad = async ({ cookies, fetch }) => {
  const refreshToken = cookies.get('refreshToken');

  if (!refreshToken) {
    return { user: null };
  }

  try {
    // 서버 사이드에서 Access Token 갱신 시도
    const response = await fetch('/api/auth/refresh', {
      method: 'POST',
      headers: { Cookie: `refreshToken=${refreshToken}` },
    });

    if (!response.ok) {
      cookies.delete('refreshToken', { path: '/' });
      return { user: null };
    }

    const { data } = await response.json();
    return {
      accessToken: data.accessToken,
      user: data.user ?? null,
    };
  } catch {
    return { user: null };
  }
};
```

### 3.4 게시글 작성 페이지 서버 액션

```typescript
// apps/frontend/src/routes/boards/[board_id]/write/+page.server.ts

import type { Actions } from './$types';
import { fail, redirect } from '@sveltejs/kit';
import { BACKEND_URL } from '$env/static/private';

export const actions: Actions = {
  default: async ({ request, params, cookies, fetch }) => {
    const formData = await request.formData();
    const title = formData.get('title') as string;
    const content = formData.get('content') as string;
    const files = formData.getAll('files') as File[];

    // 유효성 검증
    if (!title || title.length < 1) {
      return fail(400, { error: '제목을 입력해주세요' });
    }

    // Access Token 가져오기
    // 주의: PRD 설계에서는 accessToken을 httpOnly 쿠키에 저장하지 않음 — refreshToken만 httpOnly 쿠키 저장 권장 <!-- R3 수정 -->
    const accessToken = cookies.get('accessToken');
    if (!accessToken) {
      throw redirect(302, '/login');
    }

    // FormData 구성
    const postFormData = new FormData();
    postFormData.append('title', title);
    postFormData.append('content', content);
    files.forEach(file => postFormData.append('files', file));

    // API 요청
    const response = await fetch(
      `${BACKEND_URL}/api/boards/${params.board_id}/posts`,
      {
        method: 'POST',
        headers: { Authorization: `Bearer ${accessToken}` },
        body: postFormData,
      }
    );

    if (!response.ok) {
      const error = await response.json();
      return fail(response.status, { error: error.message });
    }

    const { data } = await response.json();
    throw redirect(302, `/boards/${params.board_id}/${data.postId}`);
  },
};
```
