```typescript
// apps/backend/src/prisma/prisma.service.ts
import { Injectable, OnModuleInit, OnModuleDestroy } from '@nestjs/common';
import { PrismaClient } from '@prisma/client';

@Injectable()
export class PrismaService extends PrismaClient implements OnModuleInit, OnModuleDestroy {
  constructor() {
    super({
      log: process.env.NODE_ENV === 'development' 
        ? ['query', 'info', 'warn', 'error']
        : ['error'],
    });
  }

  async onModuleInit() {
    await this.$connect();
    
    // Soft delete 미들웨어
    this.$use(async (params, next) => {
      // findMany, findFirst에서 deletedAt IS NULL 자동 필터
      if (params.action === 'findMany' || params.action === 'findFirst') {
        if (params.model && ['User', 'Post', 'Comment'].includes(params.model)) {
          params.args = {
            ...params.args,
            where: {
              ...params.args?.where,
              deletedAt: null,
            },
          };
        }
      }
      return next(params);
    });
  }

  async onModuleDestroy() {
    await this.$disconnect();
  }
}
```

### 3.2 생성된 NestJS 코드 예시

```typescript
// apps/backend/src/posts/posts.controller.ts
import {
  Controller, Get, Post, Patch, Delete,
  Param, Body, Query, UseGuards, Request,
  ParseUUIDPipe, HttpCode, HttpStatus,
} from '@nestjs/common';
import { PostsService } from './posts.service';
import { CreatePostDto } from './dto/create-post.dto';
import { UpdatePostDto } from './dto/update-post.dto';
import { PostListQueryDto } from './dto/post-list-query.dto';
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
import { OptionalJwtAuthGuard } from '../auth/guards/optional-jwt-auth.guard';
import { Public } from '../common/decorators/public.decorator';

@Controller('boards/:boardId/posts')
export class PostsController {
  constructor(private readonly postsService: PostsService) {}

  @Get()
  @UseGuards(OptionalJwtAuthGuard)
  findAll(
    @Param('boardId', ParseUUIDPipe) boardId: string,
    @Query() query: PostListQueryDto,
    @Request() req: { user?: { sub: string; role: string } },
  ) {
    return this.postsService.findAll(boardId, query, req.user);
  }

  @Get(':postId')
  @UseGuards(OptionalJwtAuthGuard)
  findOne(
    @Param('boardId', ParseUUIDPipe) boardId: string,
    @Param('postId', ParseUUIDPipe) postId: string,
    @Request() req: { user?: { sub: string; role: string } },
  ) {
    return this.postsService.findOne(boardId, postId, req.user);
  }

  @Post()
  @UseGuards(JwtAuthGuard)
  create(
    @Param('boardId', ParseUUIDPipe) boardId: string,
    @Body() dto: CreatePostDto,
    @Request() req: { user: { sub: string; role: string } },
  ) {
    return this.postsService.create(boardId, dto, req.user.sub);
  }

  @Patch(':postId')
  @UseGuards(JwtAuthGuard)
  update(
    @Param('boardId', ParseUUIDPipe) boardId: string,
    @Param('postId', ParseUUIDPipe) postId: string,
    @Body() dto: UpdatePostDto,
    @Request() req: { user: { sub: string; role: string } },
  ) {
    return this.postsService.update(boardId, postId, dto, req.user);
  }

  @Delete(':postId')
  @UseGuards(JwtAuthGuard)
  @HttpCode(HttpStatus.NO_CONTENT)
  remove(
    @Param('boardId', ParseUUIDPipe) boardId: string,
    @Param('postId', ParseUUIDPipe) postId: string,
    @Request() req: { user: { sub: string; role: string } },
  ) {
    return this.postsService.remove(boardId, postId, req.user);
  }
}
```

```typescript
// apps/backend/src/posts/posts.service.ts
import {
  Injectable, NotFoundException, ForbiddenException,
} from '@nestjs/common';
import { BoardType, Role } from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';
import { CreatePostDto } from './dto/create-post.dto';
import { UpdatePostDto } from './dto/update-post.dto';
import { PostListQueryDto } from './dto/post-list-query.dto';

interface JwtUser {
  sub: string;
  role: string;
}

@Injectable()
export class PostsService {
  constructor(private readonly prisma: PrismaService) {}

  async findAll(boardId: string, query: PostListQueryDto, user?: JwtUser) {
    const board = await this.getBoardOrThrow(boardId);
    
    const { page = 1, limit = 20, sort = 'latest', search } = query;
    const skip = (page - 1) * limit;

    const where: Record<string, unknown> = {
      boardId,
      deletedAt: null,
    };

    // INQUIRY 게시판: 비회원은 접근 불가
    if (board.boardType === BoardType.INQUIRY) {
      if (!user) throw new ForbiddenException('로그인이 필요합니다');
      if (user.role !== Role.ADMIN) {
        where['authorId'] = user.sub;
      }
    }

    if (search) {
      where['OR'] = [
        { title: { contains: search, mode: 'insensitive' } },
        { content: { contains: search, mode: 'insensitive' } },
      ];
    }

    const orderBy = sort === 'popular'
      ? [{ viewCount: 'desc' as const }]
      : [{ isPinned: 'desc' as const }, { createdAt: 'desc' as const }];

    const [items, total] = await Promise.all([
      this.prisma.post.findMany({
        where,
        orderBy,
        skip,
        take: limit,
        select: {
          id: true,
          title: true,
          content: true,
          viewCount: true,
          isPinned: true,
          isAnonymous: true,
          createdAt: true,
          author: {
            select: { id: true, nickname: true },
          },
          board: {
            select: { id: true, name: true, boardType: true },
          },
          _count: {
            select: { comments: true, likes: true },
          },
        },
      }),
      this.prisma.post.count({ where }),
    ]);

    // 익명 게시글 처리: isAnonymous=true이고 본인이 아니면 작성자 숨김
    const processedItems = items.map(item => ({
      ...item,
      author: (item.isAnonymous && user?.sub !== item.author.id)
        ? { id: null, nickname: '익명' }
        : item.author,
      commentCount: item._count.comments,
      likeCount: item._count.likes,
      _count: undefined,
    }));

    return {
      items: processedItems,
      meta: {
        total,
        page,
        limit,
        totalPages: Math.ceil(total / limit),
      },
    };
  }

  async findOne(boardId: string, postId: string, user?: JwtUser) {
    const post = await this.prisma.post.findFirst({
      where: { id: postId, boardId, deletedAt: null },
      include: {
        author: { select: { id: true, nickname: true } },
        board: { select: { id: true, name: true, boardType: true } },
        files: true,
        tags: { include: { tag: true } },
        _count: { select: { comments: true, likes: true } },
      },
    });

    if (!post) throw new NotFoundException('게시글을 찾을 수 없습니다');

    // INQUIRY 게시판 접근 제어
    if (post.board.boardType === BoardType.INQUIRY) {
      if (!user) throw new ForbiddenException('로그인이 필요합니다');
      if (user.role !== Role.ADMIN && post.author.id !== user.sub) {
        throw new ForbiddenException('접근 권한이 없습니다');
      }
    }

    // 조회수 증가 (비동기)
    this.prisma.post.update({
      where: { id: postId },
      data: { viewCount: { increment: 1 } },
    }).catch(() => {}); // 실패해도 무시

    return {
      ...post,
      author: (post.isAnonymous && user?.sub !== post.author.id)
        ? { id: null, nickname: '익명' }
        : post.author,
      commentCount: post._count.comments,
      likeCount: post._count.likes,
    };
  }

  async create(boardId: string, dto: CreatePostDto, userId: string) {
    const board = await this.getBoardOrThrow(boardId);
    
    // NOTICE, FAQ 게시판은 ADMIN만 작성
    const user = await this.prisma.user.findUnique({ where: { id: userId } });
    if (!user) throw new ForbiddenException();
    
    if (
      [BoardType.NOTICE, BoardType.FAQ].includes(board.boardType) &&
      user.role !== Role.ADMIN
    ) {
      throw new ForbiddenException('관리자만 작성할 수 있는 게시판입니다');
    }

    const post = await this.prisma.post.create({
      data: {
        title: dto.title,
        content: dto.content,
        isAnonymous: dto.isAnonymous ?? false,
        boardId,
        authorId: userId,
        ...(dto.fileIds?.length
          ? { files: { connect: dto.fileIds.map(id => ({ id })) } }
          : {}),
      },
      include: {
        author: { select: { id: true, nickname: true } },
        board: { select: { id: true, name: true, boardType: true } },
        files: true,
      },
    });

    return post;
  }

  async remove(boardId: string, postId: string, user: JwtUser) {
    const post = await this.prisma.post.findFirst({
      where: { id: postId, boardId, deletedAt: null },
    });

    if (!post) throw new NotFoundException('게시글을 찾을 수 없습니다');

    // 작성자 또는 ADMIN만 삭제 가능
    if (post.authorId !== user.sub && user.role !== Role.ADMIN) {
      throw new ForbiddenException('삭제 권한이 없습니다');
    }

    await this.prisma.post.update({
      where: { id: postId },
      data: { deletedAt: new Date() },
    });
  }

  private async getBoardOrThrow(boardId: string) {
    const board = await this.prisma.board.findUnique({
      where: { id: boardId, isActive: true },
    });
    if (!board) throw new NotFoundException('게시판을 찾을 수 없습니다');
    return board;
  }
}
```

## 4. 유스케이스에서 통합 테스트 자동 생성

```
> @docs/USECASE.md 의 UC-001(게시글 작성)을 읽고

  테스트 케이스:
  1. 로그인한 USER가 GENERAL 게시판에 게시글 작성 성공 -> 201
  2. 비로그인 상태에서 게시글 작성 시도 -> 401
  3. USER가 NOTICE 게시판에 게시글 작성 시도 -> 403
  4. 제목 없이 게시글 작성 시도 -> 400
  5. ADMIN이 NOTICE 게시판에 게시글 작성 성공 -> 201
  6. INQUIRY 게시판 게시글을 비작성자가 조회 -> 403
  7. INQUIRY 게시판 게시글을 ADMIN이 조회 -> 200
```

```
[단계 2] 미디어 게시판 (GALLERY, VIDEO)

GALLERY: 이미지 필수 첨부 검증
```
