## 2. ERD → Prisma 스키마 자동 생성 태스크

### 2.1 완전한 Prisma 스키마 생성 태스크

```
"docs/ERD.md를 읽고 prisma/schema.prisma를 완성해줘.

[변환 규칙]
1. ERD의 각 엔티티 → Prisma model
2. ERD의 필드 타입 변환:
   int → Int
   varchar → String (@db.VarChar(n))
   text → String (@db.Text)
   datetime → DateTime
   boolean → Boolean
   enum → 별도 enum 타입 정의

3. 제약조건 변환:
   PK → @id
   UK → @unique
   FK → @relation
   NOT NULL → 기본값 (nullable이면 ? 추가)

4. Soft Delete 패턴:
   deletedAt DateTime? 모든 주요 모델에 포함

5. 타임스탬프:
   createdAt DateTime @default(now())
   updatedAt DateTime @updatedAt

6. 성능을 위한 인덱스:
   자주 조회하는 컬럼 조합에 @@index 추가

[마이그레이션]
스키마 완성 후:
pnpm prisma migrate dev --name init_schema
pnpm prisma generate

[검증]
pnpm prisma validate 실행 후 오류가 없어야 함

[성공 기준]
- pnpm prisma validate 성공
- pnpm prisma generate 성공  
- 모든 ERD 엔티티가 Prisma 모델로 변환됨
- Enum이 올바르게 정의됨
- 관계가 올바르게 설정됨"
```

### 2.2 완성된 Prisma 스키마 (board-hub 전체)

```prisma
// prisma/schema.prisma
// board-hub 멀티 게시판 허브 데이터베이스 스키마
// 생성일: 2025-06-05
// 기반: docs/ERD.md

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// ==================== Enums ====================

/**
 * 사용자 역할
 * USER: 일반 회원가입 사용자
 * ADMIN: 시스템 관리자 (게시판 관리, 사용자 관리 권한)
 */
enum Role {
  USER
  ADMIN
}

/**
 * 게시판 타입
 * 각 타입별 특수 기능이 다름
 */
enum BoardType {
  GENERAL  // 일반 게시판: 텍스트 중심
  GALLERY  // 갤러리: 이미지 + 썸네일 자동 생성
  VIDEO    // 동영상: 비디오 업로드
  QNA      // Q&A: 질문·답변·채택 기능
  NOTICE   // 공지: ADMIN 전용 작성, 상단 고정 가능
  FAQ      // FAQ: 카테고리·태그 분류
  PRIVATE  // 비공개문의: 작성자·ADMIN만 열람
  FREE     // 자유: 익명 선택 가능
}

// ==================== Models ====================

/**
 * 사용자
 * Soft Delete 적용: deletedAt이 null이면 활성 계정
 */
model User {
  id        Int       @id @default(autoincrement())
  email     String    @unique @db.VarChar(255)
  password  String    // bcrypt 해시 (salt 12)
  username  String    @unique @db.VarChar(20)
  role      Role      @default(USER)
  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime? // Soft Delete

  // Relations
  posts    Post[]
  comments Comment[]
  files    File[]

  // Indexes
  @@index([email])
  @@index([deletedAt])
  @@index([role])
  @@map("users")
}

/**
 * 게시판
 * 8종 게시판 타입을 지원
 */
model Board {
  id          Int       @id @default(autoincrement())
  name        String    @unique @db.VarChar(50)
  type        BoardType
  description String?   @db.VarChar(200)
  isActive    Boolean   @default(true)
  sortOrder   Int       @default(0) // 게시판 목록 표시 순서
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt
  deletedAt   DateTime?

  // Relations
  posts      Post[]
  categories BoardCategory[]

  // Indexes
  @@index([type])
  @@index([isActive, sortOrder])
  @@index([deletedAt])
  @@map("boards")
}

/**
 * 게시물
 * 모든 게시판 타입의 게시물을 단일 테이블에서 관리
 * 타입별 특수 필드는 nullable로 처리
 */
model Post {
  id           Int       @id @default(autoincrement())
  boardId      Int
  authorId     Int
  title        String    @db.VarChar(200)
  content      String    @db.Text // HTML 형식
  viewCount    Int       @default(0)
  commentCount Int       @default(0) // 캐시 카운터 (실제 댓글 수와 동기화 필요)
  isPinned     Boolean   @default(false) // NOTICE 타입 전용
  isAnonymous  Boolean   @default(false) // FREE 타입 전용
  thumbnailUrl String?   // GALLERY 타입 전용 (파일 업로드 후 설정)
  createdAt    DateTime  @default(now())
  updatedAt    DateTime  @updatedAt
  deletedAt    DateTime?

  // Relations
  board    Board     @relation(fields: [boardId], references: [id])
  author   User      @relation(fields: [authorId], references: [id])
  comments Comment[]
  files    File[]
  tags     PostTag[]

  // Indexes
  // 게시판별 목록 조회 (최신순, Soft Delete 필터)
  @@index([boardId, deletedAt, createdAt(sort: Desc)])
  // 공지 고정 게시물 우선 정렬
  @@index([boardId, isPinned, createdAt(sort: Desc)])
  // 작성자별 조회
  @@index([authorId, deletedAt])
  // viewCount 기반 정렬 (인기순)
  @@index([boardId, viewCount(sort: Desc)])
  @@map("posts")
}

/**
 * 댓글
 */
model Comment {
  id         Int       @id @default(autoincrement())
  postId     Int
  authorId   Int
  content    String    @db.VarChar(2000)
  isAccepted Boolean   @default(false) // QNA 타입 전용: 채택된 답변
  createdAt  DateTime  @default(now())
  updatedAt  DateTime  @updatedAt
  deletedAt  DateTime?

  // Relations
  post   Post @relation(fields: [postId], references: [id])
  author User @relation(fields: [authorId], references: [id])

  // Indexes
  @@index([postId, deletedAt, createdAt])
  @@index([authorId])
  @@index([isAccepted])
  @@map("comments")
}

/**
 * 업로드 파일
 * 이미지는 썸네일 자동 생성 (sharp)
 */
model File {
  id           Int       @id @default(autoincrement())
  postId       Int?      // nullable: 게시물 미첨부 파일도 존재 가능
  uploaderId   Int
  originalUrl  String    // /uploads/originals/uuid.jpg
  thumbnailUrl String?   // /uploads/thumbs/uuid.jpg (이미지만)
  mimeType     String    // image/jpeg, video/mp4 등
  fileSize     Int       // bytes
  originalName String    // 사용자가 업로드한 원본 파일명
  createdAt    DateTime  @default(now())
  deletedAt    DateTime?

  // Relations
  post     Post? @relation(fields: [postId], references: [id])
  uploader User  @relation(fields: [uploaderId], references: [id])

  // Indexes
  @@index([postId])
  @@index([uploaderId])
  @@map("files")
}

/**
 * 게시판 카테고리
 * FAQ 게시판의 카테고리 분류
 */
model BoardCategory {
  id        Int    @id @default(autoincrement())
  boardId   Int
  name      String @db.VarChar(50)
  sortOrder Int    @default(0)

  // Relations
  board Board @relation(fields: [boardId], references: [id])

  // Indexes
  @@index([boardId, sortOrder])
  @@map("board_categories")
}

/**
 * 게시물 태그
 * FAQ 게시판 태그 기능
 */
model PostTag {
  id     Int    @id @default(autoincrement())
  postId Int
  tag    String @db.VarChar(30)

  // Relations
  post Post @relation(fields: [postId], references: [id])

  // Indexes
  @@unique([postId, tag]) // 같은 게시물에 중복 태그 방지
  @@index([tag])
  @@map("post_tags")
}
```

## 3. PrismaService 코드 (Soft Delete 미들웨어 포함)

```typescript
// apps/backend/src/prisma/prisma.service.ts

import { Injectable, OnModuleDestroy, OnModuleInit } from '@nestjs/common';
import { PrismaClient } from '@prisma/client';

/**
 * PrismaService
 * 
 * 싱글톤 Prisma 클라이언트 서비스.
 * Soft Delete 미들웨어가 내장되어 있어,
 * 삭제된 레코드(deletedAt != null)는 일반 조회에서 자동으로 제외된다.
 */
@Injectable()
export class PrismaService extends PrismaClient implements OnModuleInit, OnModuleDestroy {
  /** Soft Delete가 적용된 모델 목록 */
  private readonly softDeleteModels = ['User', 'Board', 'Post', 'Comment', 'File'];
  
  /** Soft Delete 필터가 적용되는 Prisma 작업 */
  private readonly filteredActions = [
    'findFirst',
    'findFirstOrThrow',
    'findMany',
    'findUnique',
    'findUniqueOrThrow',
    'count',
    'aggregate',
    'groupBy',
  ];

  constructor() {
    super({
      log: process.env.NODE_ENV === 'development' 
        ? ['query', 'warn', 'error'] 
        : ['warn', 'error'],
    });

    // Soft Delete는 withSoftDelete()로 확장 — PrismaModule에서 호출
  }

  /**
   * 데이터베이스 연결 초기화
   */
  async onModuleInit(): Promise<void> {
    await this.$connect();
  }

  /**
   * 데이터베이스 연결 종료
   */
  async onModuleDestroy(): Promise<void> {
    await this.$disconnect();
  }

  /**
   * Soft Delete 미들웨어 등록
   * 
   * deletedAt이 있는 모델에서 조회 시 자동으로 
   * WHERE deletedAt IS NULL 조건을 추가한다.
   * 
   * 예외: where 절에 deletedAt이 명시적으로 포함된 경우 적용하지 않음
   */
  /**
   * Soft Delete $extends 확장 반환
   *
   * Prisma v5에서 $use는 deprecated되었으므로 $extends()를 사용한다.
   * 조회 시 deletedAt IS NULL 조건을 자동으로 추가한다.
   */
  withSoftDelete() {
    return this.$extends({
      query: {
        $allModels: {
          async $allOperations({ model, operation, args, query }: any) {
            const softDeleteModels = ['User', 'Board', 'Post', 'Comment', 'File'];
            const filteredActions = [
              'findFirst', 'findFirstOrThrow', 'findMany',
              'findUnique', 'findUniqueOrThrow', 'count', 'aggregate', 'groupBy',
            ];
            if (
              model && softDeleteModels.includes(model) &&
              filteredActions.includes(operation)
            ) {
              const where = args?.where;
              if (!where || !('deletedAt' in where)) {
                args = { ...args, where: { ...where, deletedAt: null } };
              }
            }
            return query(args);
          },
        },
      },
    });
  }

  /**
   * Soft Delete 실행
   * 
   * @param model - Prisma 모델명
   * @param where - 삭제 조건
   * @returns 업데이트된 레코드
   */
  async softDelete<T>(
    model: keyof PrismaClient,
    where: Record<string, unknown>,
  ): Promise<T> {
    // TODO(R3): Function 타입은 TypeScript 타입 안전성을 저해합니다.
    // 실제 프로덕션에서는 Prisma 모델 delegate 타입을 명시하거나
    // $extends() 방식으로 Soft Delete를 전면 구현하는 것을 권장합니다.
    const delegate = this[model] as { update: Function };
    return delegate.update({
      where,
      data: { deletedAt: new Date() },
    });
  }
}
```

### 4.2 UserModule 예시

```typescript
// apps/backend/src/user/user.service.ts

import {
  Injectable,
  ConflictException,
  NotFoundException,
} from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { Role, User } from '@prisma/client';
import * as bcrypt from 'bcrypt';

@Injectable()
export class UserService {
  constructor(private readonly prisma: PrismaService) {}

  /**
   * 이메일로 사용자 조회 (비밀번호 포함)
   * 인증 목적으로만 사용한다.
   * 
   * @param email - 조회할 이메일 주소
   * @returns 사용자 또는 null
   */
  async findByEmailWithPassword(email: string): Promise<User | null> {
    return this.prisma.user.findUnique({
      where: { email },
    });
  }

  /**
   * ID로 사용자 조회 (비밀번호 제외)
   * 
   * @param id - 사용자 ID
   * @returns 사용자 (비밀번호 제외)
   * @throws NotFoundException - 사용자가 존재하지 않는 경우
   */
  async findById(id: number): Promise<Omit<User, 'password'>> {
    const user = await this.prisma.user.findUnique({
      where: { id },
      select: {
        id: true,
        email: true,
        username: true,
        role: true,
        createdAt: true,
        updatedAt: true,
        deletedAt: true,
        password: false,
      },
    });

    if (!user) {
      throw new NotFoundException('사용자를 찾을 수 없습니다');
    }

    return user;
  }

  /**
   * 회원가입
   * 
   * @param email - 이메일 주소
   * @param password - 평문 비밀번호 (내부에서 해싱)
   * @param username - 사용자명
   * @returns 생성된 사용자 (비밀번호 제외)
   * @throws ConflictException - 이메일 또는 사용자명 중복
   */
  async create(
    email: string,
    password: string,
    username: string,
  ): Promise<Omit<User, 'password'>> {
    // 중복 확인
    const [emailExists, usernameExists] = await Promise.all([
      this.prisma.user.findUnique({ where: { email } }),
      this.prisma.user.findUnique({ where: { username } }),
    ]);

    if (emailExists) {
      throw new ConflictException('이미 사용 중인 이메일입니다');
    }
    if (usernameExists) {
      throw new ConflictException('이미 사용 중인 사용자명입니다');
    }

    // 비밀번호 해싱
    const hashedPassword = await bcrypt.hash(password, 12);

    // 사용자 생성
    const user = await this.prisma.user.create({
      data: {
        email,
        password: hashedPassword,
        username,
        role: Role.USER,
      },
    });

    // 비밀번호 제외하고 반환
    const { password: _, ...userWithoutPassword } = user;
    return userWithoutPassword;
  }
}
```

@ApiTags('Posts')
@Controller()
export class PostController {
  constructor(private readonly postService: PostService) {}

  /**
   * GET /api/v1/boards/:boardId/posts
   * 게시물 목록 조회 (페이지네이션)
   */
  @Get('boards/:boardId/posts')
  @UseGuards(OptionalJwtGuard)
  @ApiOperation({ summary: '게시물 목록 조회' })
  @ApiQuery({ name: 'page', required: false, type: Number, example: 1 })
  @ApiQuery({ name: 'limit', required: false, type: Number, example: 20 })
  @ApiQuery({ name: 'sort', required: false, enum: ['createdAt', 'viewCount'] })
  @ApiResponse({ status: 200, description: '게시물 목록' })
  @ApiResponse({ status: 403, description: 'PRIVATE 게시판 - 비로그인' })
  @ApiResponse({ status: 404, description: '게시판 미존재' })
  async findAll(
    @Param('boardId', ParseIntPipe) boardId: number,
    @Query('page') page = 1,
    @Query('limit') limit = 20,
    @Query('sort') sort: 'createdAt' | 'viewCount' = 'createdAt',
    @Request() req: any,
  ) {
    return this.postService.findAll({
      boardId,
      page: Number(page),
      limit: Math.min(Number(limit), 100),
      sort,
      userId: req.user?.id,
      userRole: req.user?.role,
    });
  }

  /**
   * GET /api/v1/posts/:id
   * 게시물 상세 조회 (viewCount +1)
   */
  @Get('posts/:id')
  @UseGuards(OptionalJwtGuard)
  @ApiOperation({ summary: '게시물 상세 조회' })
  @ApiResponse({ status: 200, description: '게시물 상세' })
  @ApiResponse({ status: 403, description: 'PRIVATE 게시판 접근 불가' })
  @ApiResponse({ status: 404, description: '게시물 미존재' })
  async findOne(
    @Param('id', ParseIntPipe) id: number,
    @Request() req: any,
  ) {
    return this.postService.findOne(id, req.user?.id, req.user?.role);
  }

  /**
   * POST /api/v1/posts
   * 게시물 작성
   */
  @Post('posts')
  @UseGuards(JwtAuthGuard)
  @HttpCode(HttpStatus.CREATED)
  @ApiBearerAuth()
  @ApiOperation({ summary: '게시물 작성' })
  @ApiResponse({ status: 201, description: '게시물 생성 성공' })
  @ApiResponse({ status: 401, description: '미인증' })
  @ApiResponse({ status: 403, description: 'NOTICE 게시판 - USER 권한' })
  async create(
    @Body() createPostDto: CreatePostDto,
    @Request() req: any,
  ) {
    return this.postService.create(createPostDto, req.user.id, req.user.role);
  }

  /**
   * PATCH /api/v1/posts/:id
   * 게시물 수정 (본인 또는 ADMIN)
   */
  @Patch('posts/:id')
  @UseGuards(JwtAuthGuard)
  @ApiBearerAuth()
  @ApiOperation({ summary: '게시물 수정' })
  async update(
    @Param('id', ParseIntPipe) id: number,
    @Body() updatePostDto: UpdatePostDto,
    @Request() req: any,
  ) {
    return this.postService.update(id, updatePostDto, req.user.id, req.user.role);
  }

  /**
   * DELETE /api/v1/posts/:id
   * 게시물 삭제 (Soft Delete)
   */
  @Delete('posts/:id')
  @UseGuards(JwtAuthGuard)
  @HttpCode(HttpStatus.NO_CONTENT)
  @ApiBearerAuth()
  @ApiOperation({ summary: '게시물 삭제 (소프트 삭제)' })
  async remove(
    @Param('id', ParseIntPipe) id: number,
    @Request() req: any,
  ) {
    return this.postService.remove(id, req.user.id, req.user.role);
  }

  /**
   * PATCH /api/v1/posts/:id/pin
   * 공지 게시물 상단 고정 (ADMIN 전용)
   */
  @Patch('posts/:id/pin')
  @UseGuards(JwtAuthGuard)
  @ApiBearerAuth()
  @ApiOperation({ summary: '게시물 고정/해제 (ADMIN 전용)' })
  async togglePin(
    @Param('id', ParseIntPipe) id: number,
    @Request() req: any,
  ) {
    return this.postService.togglePin(id, req.user.role);
  }
}
