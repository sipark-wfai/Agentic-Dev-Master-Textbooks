### 2.2 JWT 인증 스토어

```typescript
// apps/frontend/src/lib/stores/auth.ts

import { writable, derived } from 'svelte/store';
import type { Writable } from 'svelte/store';

/** 사용자 정보 타입 */
export interface AuthUser {
  id: number;
  email: string;
  username: string;
  role: 'USER' | 'ADMIN';
}

/** 인증 상태 */
interface AuthState {
  user: AuthUser | null;
  isLoading: boolean;
}

/** 인증 상태 스토어 */
const authStore: Writable<AuthState> = writable({
  user: null,
  isLoading: true,
});

/** 사용자 파생 스토어 */
export const user = derived(authStore, ($auth) => $auth.user);

/** 로그인 여부 파생 스토어 */
export const isAuthenticated = derived(
  authStore,
  ($auth) => $auth.user !== null,
);

/** ADMIN 여부 파생 스토어 */
export const isAdmin = derived(
  authStore,
  ($auth) => $auth.user?.role === 'ADMIN',
);

/** 로딩 여부 파생 스토어 */
export const isLoading = derived(authStore, ($auth) => $auth.isLoading);

/**
 * 사용자 정보 설정 (로그인 후 호출)
 */
export function setUser(userData: AuthUser): void {
  authStore.update((state) => ({
    ...state,
    user: userData,
    isLoading: false,
  }));
}

/**
 * 사용자 정보 초기화 (로그아웃 후 호출)
 */
export function clearUser(): void {
  authStore.update((state) => ({
    ...state,
    user: null,
    isLoading: false,
  }));
}

/**
 * 로딩 상태 설정
 */
export function setLoading(loading: boolean): void {
  authStore.update((state) => ({
    ...state,
    isLoading: loading,
  }));
}
```

```typescript
// apps/frontend/src/routes/+layout.server.ts

import type { LayoutServerLoad } from './$types';

/**
 * 전체 레이아웃 서버 로드
 * 모든 페이지에서 사용자 정보를 확인한다.
 */
export const load: LayoutServerLoad = async ({ fetch, cookies }) => {
  const token = cookies.get('access_token');
  
  if (!token) {
    return { user: null };
  }

  try {
    const response = await fetch('/api/v1/auth/me');
    
    if (!response.ok) {
      return { user: null };
    }
    
    const user = await response.json();
    return { user };
  } catch {
    return { user: null };
  }
};
```

### 3.1 게시판 목록 페이지 (+page.server.ts)

```typescript
// apps/frontend/src/routes/board/[board_id]/+page.server.ts

import type { PageServerLoad, Actions } from './$types';
import { error } from '@sveltejs/kit';

export const load: PageServerLoad = async ({ params, fetch, url }) => {
  const boardId = Number(params.board_id);
  const page = Number(url.searchParams.get('page') ?? '1');
  const limit = Number(url.searchParams.get('limit') ?? '20');
  const sort = url.searchParams.get('sort') ?? 'createdAt';

  // 게시판 정보 조회
  const boardRes = await fetch(`/api/v1/boards/${boardId}`);
  if (!boardRes.ok) {
    throw error(boardRes.status, '게시판을 찾을 수 없습니다');
  }
  const board = await boardRes.json();

  // 게시물 목록 조회
  const postsRes = await fetch(
    `/api/v1/boards/${boardId}/posts?page=${page}&limit=${limit}&sort=${sort}`
  );
  if (!postsRes.ok) {
    throw error(postsRes.status, '게시물 목록을 불러올 수 없습니다');
  }
  const { posts, pagination } = await postsRes.json();

  return { board, posts, pagination };
};
```

### 4.1 로그인 Form Action

```typescript
// apps/frontend/src/routes/auth/login/+page.server.ts

import type { Actions, PageServerLoad } from './$types';
import { fail, redirect } from '@sveltejs/kit';

export const load: PageServerLoad = async ({ locals }) => {
  // 이미 로그인된 경우 홈으로
  if (locals.user) {
    throw redirect(302, '/');
  }
  return {};
};

export const actions: Actions = {
  default: async ({ request, fetch, cookies }) => {
    const data = await request.formData();
    const email = data.get('email')?.toString();
    const password = data.get('password')?.toString();

    if (!email || !password) {
      return fail(400, { message: '이메일과 비밀번호를 입력해주세요' });
    }

    try {
      const response = await fetch('/api/v1/auth/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password }),
        credentials: 'include',
      });

      if (!response.ok) {
        const error = await response.json();
        return fail(response.status, {
          message: error.message ?? '로그인에 실패했습니다',
        });
      }

      // JWT 쿠키가 응답에 포함됨 (백엔드에서 Set-Cookie)
      throw redirect(302, '/');
    } catch (e) {
      if (e instanceof Response) throw e;
      return fail(500, { message: '서버 오류가 발생했습니다' });
    }
  },
};
```

```typescript
  app.enableCors({
    origin: process.env.FRONTEND_URL || 'http://localhost:5173',
    credentials: true, // 쿠키 전송 허용
    methods: ['GET', 'POST', 'PUT', 'PATCH', 'DELETE', 'OPTIONS'],
  });

  // 글로벌 API 접두사
  app.setGlobalPrefix('api/v1');

  // 글로벌 유효성 검사 파이프
  app.useGlobalPipes(
    new ValidationPipe({
      whitelist: true,       // DTO에 없는 속성 제거
      forbidNonWhitelisted: true, // DTO에 없는 속성 있으면 오류
      transform: true,       // 타입 자동 변환
      transformOptions: {
        enableImplicitConversion: true,
      },
    }),
  );

  // Swagger 설정 (개발 환경만)
  if (process.env.NODE_ENV !== 'production') {
    const config = new DocumentBuilder()
      .setTitle('Board Hub API')
      .setDescription('멀티 게시판 허브 REST API')
      .setVersion('1.0')
      .addCookieAuth('access_token')
      .build();
    
    const document = SwaggerModule.createDocument(app, config);
    SwaggerModule.setup('api/docs', app, document);
    logger.log('Swagger UI: http://localhost:3000/api/docs');
  }

  const port = process.env.PORT || 3000;
  await app.listen(port);
  logger.log(`Application is running on: http://localhost:${port}`);
}

bootstrap();
```
